Look at a single lion:

def count_operations(node, level=0, counts=None):
  if counts is None:
    counts = build_counts()
  op = node['op']
  counts[op][level] += 1
  for child in node["children"]: 
    count_operations(child, level+1, counts)
  return counts

def plot_depths(gens, dfs, ax, title_specs):
